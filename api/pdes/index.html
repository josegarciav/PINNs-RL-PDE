    PDEs¶

    Every shipped PDE is registered once in pdes.registry, and both the CLI and the dashboard read it from there. User-defined equations go through CustomPDE (see Custom equations).

    def __init__(
    self,
    equations: Equations,
    *,
    domain: Mapping[str, Sequence[float]],
    fields: Sequence[str] = ("u",),
    parameters: Mapping[str, float] | None = None,
    initial_condition: Any = None,
    initial_velocity: Any = None,
    boundary_conditions: Mapping[str, Mapping[str, Any]] | None = None,
    exact_solution: Any = None,
    name: str = "Custom PDE",
    trainable_parameters: Sequence[str] = (),
    parameter_initial_guesses: Mapping[str, float] | None = None,
    observation_data: Mapping[str, Any] | None = None,
    training: Any = None,
    device: torch.device | str | None = None,
    time_order: int | None = None,
    architecture: str = "fourier",
    **_ignored: Any,
):
    """Build the problem; see the class docstring for the arguments."""
    self.fields = tuple(fields)
    if not self.fields:
        raise ValueError("fields must name at least one network output")
    domain = {str(k): (float(v[0]), float(v[1])) for k, v in domain.items()}
    self.spatial_coordinates = tuple(k for k in domain if k != "t")
    self.time_dependent = "t" in domain
    self.is_ode = not self.spatial_coordinates
    if self.is_ode:
        self.spatial_coordinates = (_DUMMY_COORDINATE,)
        domain = {_DUMMY_COORDINATE: (0.0, 1.0), **domain}
    self.coordinates = self.spatial_coordinates + (("t",) if self.time_dependent else ())
    self.parameters = {str(k): float(v) for k, v in (parameters or {}).items()}
    self._architecture = architecture
    self._spec = {
        "equations": equations if not callable(equations) else None,
        "domain": {k: list(v) for k, v in domain.items() if k != _DUMMY_COORDINATE},
        "fields": list(self.fields),
        "parameters": dict(self.parameters),
        "initial_condition": None if callable(initial_condition) else initial_condition,
        "initial_velocity": None if callable(initial_velocity) else initial_velocity,
        "boundary_conditions": copy.deepcopy(dict(boundary_conditions or {})),
        "exact_solution": None if callable(exact_solution) else exact_solution,
        "name": name,
        "time_order": time_order,
        "architecture": architecture,
    }

    # Residual: compiled expressions or the user's callable.
    self._residual_fn: Callable | None = None
    self._equations: list[CompiledExpression] = []
    if callable(equations):
        self._residual_fn = equations
        self._needed: set[str] = set()
    else:
        texts = [equations] if isinstance(equations, str) else list(equations)
        if len(texts) != len(self.fields):
            raise ValueError(
                f"{len(texts)} equation(s) for {len(self.fields)} field(s) {self.fields}; "
                "give one equation per field"
            )
        self._equations = [
            compile_expression(t, self.fields, self.coordinates, self.parameters) for t in texts
        ]
        self._needed = {p for e in self._equations for p in e.parameters}
    self._ic = self._compile_map(initial_condition, "initial_condition", with_time=False)
    self._ic_fn = initial_condition if callable(initial_condition) else None
    self._velocity = self._compile_map(initial_velocity, "initial_velocity", with_time=False)
    self._velocity_fn = initial_velocity if callable(initial_velocity) else None
    self._exact = self._compile_map(exact_solution, "exact_solution", with_time=True)
    self._exact_fn = exact_solution if callable(exact_solution) else None
    for compiled in list((self._ic or {}).values()) + list((self._velocity or {}).values()):
        self._needed |= set(compiled.parameters)
    if self._exact:
        self._needed |= {p for e in self._exact.values() for p in e.parameters}
    missing = sorted(self._needed - set(self.parameters))
    if missing:
        raise ValueError(
            f"Parameters {missing} appear in the equations but are not in parameters="
            f"{sorted(self.parameters)}; pass a value for each (or list it in "
            "trainable_parameters with an initial guess)"
        )

    if time_order is None:
        orders = [e.time_order for e in self._equations]
        time_order = max(orders, default=1) if self.time_dependent else 0
    self.time_order = max(int(time_order), 1) if self.time_dependent else 0

    has_ic = initial_condition is not None and self.time_dependent
    config = PDEConfig(
        name=name,
        domain=[domain[c] for c in self.spatial_coordinates],
        time_domain=domain["t"] if self.time_dependent else (0.0, 0.0),
        parameters=dict(self.parameters),
        boundary_conditions=copy.deepcopy(dict(boundary_conditions or {})),
        initial_condition={"type": "custom"} if has_ic else {},
        exact_solution={"type": "custom"} if exact_solution is not None else {},
        dimension=len(self.spatial_coordinates),
        input_dim=len(self.spatial_coordinates) + 1,
        output_dim=len(self.fields),
        architecture=architecture,
        device=torch.device(device) if device is not None else None,
        training=training,
        trainable_parameters=list(trainable_parameters),
        parameter_initial_guesses=dict(parameter_initial_guesses or {}),
        observation_data=dict(observation_data) if observation_data else None,
    )
    super().__init__(config)

def from_dict(cls, spec: Mapping[str, Any], **overrides: Any) -> CustomPDE:
    """Build from a plain dict (the YAML layout of :meth:`to_dict`); ``overrides`` win."""
    spec = {**spec, **overrides}

    return cls(equations, **spec)

    def build_trainer(
    self,
    architecture: str | None = None,
    config_path: str | None = None,
    device: torch.device | str | None = None,
    model: Mapping[str, Any] | None = None,
    **training: Any,
):
    """A :class:`~pinnrl.training.trainer.PDETrainer` for this problem with packaged defaults.

    ``architecture`` names an entry of the ``architectures:`` block of
    ``config.yaml`` (``config_path`` for your own file); ``model``
    overrides keys of that entry (``hidden_dims``, ``activation``,
    ``layer_norm``, ``scale``, ...). ``training`` overrides keys of the
    ``training:`` block (``num_epochs``, ``batch_size``,
    ``num_collocation_points``, ``learning_rate``, ``weight_decay``,
    ``mode``, ``loss_weights``, ...).
    """
    from pinnrl.config import DEFAULT_CONFIG_PATH, Config, ModelConfig, TrainingConfig
    from pinnrl.neural_networks import PINNModel
    from pinnrl.training.trainer import PDETrainer

    with open(config_path or DEFAULT_CONFIG_PATH) as f:
        defaults = yaml.safe_load(f)
    training_cfg = copy.deepcopy(defaults.get("training", {}))
    for key, value in training.items():
        if key in ("learning_rate", "weight_decay"):
            training_cfg.setdefault("optimizer_config", {})[key] = value
        elif isinstance(value, Mapping) and isinstance(training_cfg.get(key), Mapping):
            training_cfg[key] = {**training_cfg[key], **value}
        else:
            training_cfg[key] = value
    if self.config.trainable_parameters and "mode" not in training:
        training_cfg["mode"] = "inverse"
    architecture = architecture or self._architecture
    device = torch.device(device) if device is not None else self.device
    config = Config.__new__(Config)
    config.device = device
    config.training = TrainingConfig.from_dict(training_cfg)
    config.model = ModelConfig.from_architecture(
        architecture,
        {**defaults.get("architectures", {}).get(architecture, {}), **dict(model or {})},
        input_dim=self.config.input_dim,
        output_dim=self.config.output_dim,
    )
    config.rl = None
    self.config.training = config.training
    self.device = device
    self.config.device = device
    model = PINNModel(config=config, device=device).to(device)
    return PDETrainer(model=model, pde=self, config=config, device=device)

    def compute_residual(
    self, model: torch.nn.Module, x: torch.Tensor, t: torch.Tensor
) -> torch.Tensor:
    """Residuals of every equation at ``x, t``, shape ``(N, k)``."""
    f = Fields(model, x, t, self.fields, self.coordinates, self.device)
    params = self._parameter_values()
    if self._residual_fn is not None:
        out = self._residual_fn(f, params)
        if not isinstance(out, torch.Tensor):
            out = torch.cat([o.reshape(f.n_points, -1) for o in out], dim=1)
        return out.reshape(f.n_points, -1)
    like = f.value(self.fields[0])
    columns = []
    for expr in self._equations:
        values: dict[str, Any] = dict(params)
        for field in expr.fields:
            values[field] = f.value(field)
        for field, coords in expr.derivatives:
            values[derivative_name(field, coords)] = f.d(field, *coords)
        for c in expr.coordinates:
            values[c] = f.coordinate(c)
        columns.append(expr(values, like))
    return torch.cat(columns, dim=1)

    def d(self, field: str, *coordinates: str) -> torch.Tensor:
    """``∂^n field / ∂coordinates``, e.g. ``f.d("u", "x", "x")`` for ``u_xx``."""
    key = (field, tuple(sorted(coordinates)))
    if not coordinates:
        return self.value(field)
    if key not in self._cache:
        previous = self.d(field, *key[1][:-1])
        wrt = self._t if key[1][-1] == "t" else self._x
        grad = PDEBase._grad(previous, wrt)
        if key[1][-1] != "t":
            i = self._spatial.index(key[1][-1])
            grad = grad[:, i : i + 1]
        self._cache[key] = grad
    return self._cache[key]

    def compile_expression(
    text: str,
    fields: Sequence[str],
    coordinates: Sequence[str],
    known_parameters: Iterable[str] = (),
) -> CompiledExpression:
    """Parse ``text`` and compile it into a torch callable.

    Args:
        text: The expression, e.g. ``"u_t + u*u_x - nu*u_xx"``.
        fields: Field (output) names.
        coordinates: Coordinate (input) names, including ``"t"`` when the
            problem is time dependent.
        known_parameters: Parameter names the caller can supply; any other
            free name is still accepted and reported in ``parameters`` so the
            caller can raise a precise error.

    Raises:
        ValueError: If the text does not parse or uses an unknown function.
    """
    fields = tuple(fields)
    coordinates = tuple(coordinates)
    local: dict[str, object] = {}
    seen_fields, seen_derivs, seen_coords, seen_params = [], [], [], []
    for ident in dict.fromkeys(_IDENTIFIER.findall(text)):
        canonical = _FUNCTION_ALIASES.get(ident, ident)
        if ident in fields:
            local[ident] = sympy.Symbol(ident)
            seen_fields.append(ident)
            continue
        split = split_derivative(ident, fields, coordinates)
        if split is not None:
            name = derivative_name(*split)
            local[ident] = sympy.Symbol(name)
            if split not in seen_derivs:
                seen_derivs.append(split)
            continue
        if ident in coordinates:
            local[ident] = sympy.Symbol(ident)
            seen_coords.append(ident)
        elif canonical in _SYMPY_FUNCTIONS:
            local[ident] = _SYMPY_FUNCTIONS[canonical]
        elif canonical in TORCH_FUNCTIONS:
            local[ident] = getattr(sympy, canonical)
        elif ident in _CONSTANTS:
            local[ident] = _CONSTANTS[ident]
        else:
            looks_like = [f for f in fields if ident.startswith(f + "_")]
            if looks_like:
                raise ValueError(
                    f"{ident!r} in {text!r} reads as a derivative of {looks_like[0]!r}, but "
                    f"{ident[len(looks_like[0]) + 1 :]!r} is not made of the coordinates "
                    f"{tuple(coordinates)}"
                )
            local[ident] = sympy.Symbol(ident)
            seen_params.append(ident)
    try:
        expr = parse_expr(text, local_dict=local, transformations=_TRANSFORMATIONS)
    except Exception as exc:  # SyntaxError, TokenError, TypeError from SymPy
        raise ValueError(f"Cannot parse expression {text!r}: {exc}") from exc
    if not isinstance(expr, (sympy.Expr, sympy.Basic)):
        raise ValueError(f"Expression {text!r} is not a formula")
    for func in expr.atoms(sympy.Function):
        name = type(func).__name__
        if name not in TORCH_FUNCTIONS and name not in _SYMPY_FUNCTIONS:
            raise ValueError(
                f"Unknown function {name!r} in {text!r}; available: "
                + ", ".join(sorted(k for k in TORCH_FUNCTIONS if k[0].islower()))
            )
    deriv_names = [derivative_name(f, c) for f, c in seen_derivs]
    args = tuple(seen_fields + deriv_names + seen_coords + seen_params)
    symbols = [sympy.Symbol(a) for a in args]
    fn = sympy.lambdify(symbols, expr, modules=[TORCH_FUNCTIONS, {"pi": math.pi}])
    return CompiledExpression(
        text=text,
        expr=expr,
        fields=tuple(seen_fields),
        derivatives=tuple(seen_derivs),
        coordinates=tuple(seen_coords),
        parameters=tuple(seen_params),
        _fn=fn,
        _args=args,
    )

    def __call__(self, values: dict[str, torch.Tensor | float], like: torch.Tensor) -> torch.Tensor:
    """Evaluate with ``values`` (name to tensor or number); constants broadcast to ``like``.

    Derivative names use the subscript spelling the expression used

    """
    out = self._fn(*(values[a] for a in self._args))
    if not isinstance(out, torch.Tensor):
        return torch.full_like(like, float(out))
    if out.shape != like.shape:
        out = out.expand_as(like) if out.dim() <= like.dim() else out
    return out

    PDEBase ¶

    PDEBase(config, bandit_sampler=None)

    Base class for PDEs solved by PINNs.

    A subclass provides the physics:

    • :meth:compute_residual — the PDE operator applied to the network.
    • :meth:_create_initial_condition — u(x, t0) for its IC types.
    • :meth:_exact_solution — a closed-form solution, when one exists.
    • time_order = 2 for second-order-in-time PDEs; the initial loss then also enforces u_t(x, t0) = initial_condition["velocity"] (default 0).

    The base class owns everything else: derivatives, collocation sampling, boundary conditions and the composite loss. boundary_conditions in the config maps a kind to its options; every spatial axis gets the condition on both faces:

    • dirichlet: {"value": v} or {"left": a, "right": b} (the lower / upper face), or {"type": "exact"} to take values from the exact solution;
    • neumann: {"value": g} — du/dx_axis = g on both faces;
    • periodic: {} — u and du/dx_axis agree on opposite faces.

    exact_solution: {} in the config means the problem has no closed-form solution; :meth:exact_solution then returns None.

    def __init__(self, config: PDEConfig, bandit_sampler=None):
    """Initialize PDE with given configuration.

    Args:
        config: Configuration dataclass.
        bandit_sampler: Optional :class:`ResidualBanditSampler` for the
            ``adaptive`` strategy; built lazily with defaults when omitted.
    """
    self.config = config
    self.domain = config.domain
    self.bandit_sampler = bandit_sampler

    # One (lo, hi) pair per spatial axis; a bare (lo, hi) means one axis.
    if not self.domain:
        self.domain = [(0.0, 1.0)]
    elif isinstance(self.domain[0], (list, tuple)):
        self.domain = [(float(lo), float(hi)) for lo, hi in self.domain]
    else:
        self.domain = [(float(self.domain[0]), float(self.domain[1]))]

    # Update config with normalized domain format
    self.config.domain = self.domain

    # Handle time_domain/t_domain property
    if hasattr(config, "time_domain"):
        self.time_domain = config.time_domain
    elif hasattr(config, "t_domain"):
        self.time_domain = config.t_domain
    else:
        self.time_domain = [0.0, 1.0]

    # Convert time_domain to tuple if it's a list
    if isinstance(self.time_domain, list):
        self.time_domain = tuple(self.time_domain)

    # Setup device - Handle device explicitly and carefully
    if hasattr(config, "device") and config.device is not None:
        # If config.device is already a torch.device object, use it directly
        if isinstance(config.device, torch.device):
            self.device = config.device
        else:
            # Try to convert string to torch.device
            try:
                self.device = torch.device(str(config.device))
            except Exception:
                print(f"Warning: Invalid device '{config.device}', falling back to CPU")
                self.device = torch.device("cpu")
    else:
        # Fallback to CPU if no device specified
        self.device = torch.device("cpu")

    # Ensure config has the same device
    self.config.device = self.device

    # Store dimensionality
    self.dimension = config.dimension

    # Make sure parameters exist but preserve any existing parameters
    if not hasattr(config, "parameters"):
        config.parameters = {}
    elif config.parameters is None:
        config.parameters = {}

    # Register trainable parameters as nn.Parameter (inverse-problem mode).
    # In forward mode this dict stays empty and adds no behaviour.
    self._trainable_params: nn.ParameterDict = nn.ParameterDict()
    trainable_names = list(getattr(config, "trainable_parameters", []) or [])
    initial_guesses = dict(getattr(config, "parameter_initial_guesses", {}) or {})
    # Snapshot the original (true) values BEFORE the parameters become
    # trainable, so we can plot a reference line in the dashboard.
    self._true_parameters: dict[str, float] = {}
    for name in trainable_names:
        true_val = _as_scalar(name, config.parameters.get(name))
        if true_val is not None:
            self._true_parameters[name] = true_val
        init_val = _as_scalar(
            name, initial_guesses.get(name, true_val if true_val is not None else 1.0)
        )
        self._trainable_params[name] = nn.Parameter(torch.tensor(init_val, device=self.device))

    # Load observation data (used by the data-fitting loss in inverse mode).
    self.observation_data: dict[str, torch.Tensor] | None = self._load_observation_data(
        getattr(config, "observation_data", None)
    )
    # No time window of its own (a Well dataset, whose trajectory length can
    # differ by file): span the time range of the observations, else [0, 1].
    if self.time_domain is None:
        if self.observation_data is not None:
            t = self.observation_data["t"]
            self.time_domain = (float(t.min()), float(t.max()))
        else:
            self.time_domain = (0.0, 1.0)
        self.config.time_domain = self.time_domain

    self.initial_condition = self._create_initial_condition(
        dict(config.initial_condition or {})
    )
    self.boundary_conditions = self._parse_boundary_conditions(config.boundary_conditions or {})
    self._fixed_points: dict[tuple[str, int], Any] = {}

    # Setup neural network parameters if not specified
    if self.config.input_dim is None:
        self.config.input_dim = self.dimension + 1  # Spatial dimensions + time

    if self.config.output_dim is None:
        self.config.output_dim = 1  # Default to single output (u)

    def generate_synthetic_observations(
    self,
    n_points: int = 200,
    noise_std: float = 0.0,
    seed: int | None = 0,
) -> dict[str, torch.Tensor]:
    """Sample noisy observations from the analytical solution.

    Useful for inverse-problem demos and CI tests: pick uniform points in
    the domain, evaluate ``self.exact_solution``, optionally add Gaussian
    noise, and store the result on ``self.observation_data`` so the next
    ``compute_loss`` call automatically picks it up.

    Args:
        n_points: Number of observations.
        noise_std: Standard deviation of the additive Gaussian noise.
        seed: Seed for the points and the noise; ``None`` leaves it unseeded.

    Returns:
        ``{"x", "t", "u"}`` tensors, also stored on ``self.observation_data``.

    Raises:
        ValueError: If the PDE has no exact solution configured.
    """
    gen = torch.Generator(device="cpu")
    if seed is not None:
        gen.manual_seed(int(seed))

    spatial_dims = max(int(self.dimension), 1)
    x_cols = []
    for d in range(spatial_dims):
        lo, hi = self.domain[d]
        x_cols.append(torch.rand(n_points, 1, generator=gen) * (hi - lo) + lo)
    x = (
        torch.cat(x_cols, dim=1).to(self.device)
        if spatial_dims > 1
        else x_cols[0].to(self.device)
    )

    t_lo, t_hi = self.time_domain[0], self.time_domain[1]
    t = (torch.rand(n_points, 1, generator=gen) * (t_hi - t_lo) + t_lo).to(self.device)

    # Temporarily unregister the trainable params so ``get_parameter``
    # returns the TRUE values from ``config.parameters`` (the ones
    # snapshotted in ``_true_parameters``) rather than the current guess.
    # Without this, synthetic obs generated in inverse mode would already
    # encode the wrong α and the trainer would converge to the guess
    # rather than recovering the true value.
    saved_params = self._trainable_params if hasattr(self, "_trainable_params") else None
    try:
        if saved_params is not None and self._true_parameters:
            self._trainable_params = nn.ParameterDict()
        if not self.has_exact_solution:
            raise ValueError(
                f"{self.config.name}: synthetic observations need a configured exact_solution"
            )
        with torch.no_grad():
            u = self.exact_solution(x, t)
            if noise_std and noise_std > 0:
                noise = torch.randn(u.shape, generator=gen).to(self.device) * float(noise_std)
                u = u + noise
    finally:
        if saved_params is not None:
            self._trainable_params = saved_params

    self.observation_data = {"x": x, "t": t, "u": u}
    return self.observation_data

    def laplacian(self, f: torch.Tensor, x: torch.Tensor) -> torch.Tensor:
    """Sum of the pure second derivatives of ``f`` along each column of ``x``.

    ``x`` must be the leaf tensor ``f`` was computed from (``requires_grad``).
    Differentiating with respect to the whole tensor and then taking a column
    is what makes this correct in more than one dimension; a column slice
    ``x[:, d:d+1]`` is not part of the graph.
    """
    grad = self._grad(f, x)
    lap = torch.zeros_like(f)
    for d in range(x.shape[1]):
        lap = lap + self._grad(grad[:, d : d + 1], x)[:, d : d + 1]
    return lap

        ValueError: For temporal orders above 2 or spatial orders above 4.
    """
    temporal = sorted(set(temporal_derivatives or ()) - {0})
    spatial = sorted(set(spatial_derivatives or ()) - {0})
    if temporal and temporal[-1] > 2:
        raise ValueError(
            f"Temporal derivative order {temporal[-1]} is not supported. Maximum order is 2."
        )
    if spatial and spatial[-1] > 4:
        raise ValueError(
            f"Spatial derivative order {spatial[-1]} is not supported. Maximum order is 4."
        )

    x, t, u = self._prepare_inputs(model, x, t)
    derivatives = {"u": u}

    current = u
    for order in range(1, (temporal[-1] if temporal else 0) + 1):
        current = self._grad(current, t)
        if order in temporal:
            derivatives["dt" if order == 1 else f"dt{order}"] = current

    for dim in range(self.dimension):
        current = u
        for order in range(1, (spatial[-1] if spatial else 0) + 1):
            current = self._grad(current, x)[:, dim : dim + 1]
            if order not in spatial:
                continue
            if self.dimension == 1:
                key = "dx" if order == 1 else f"dx{order}"
            else:
                key = "d" + f"x{dim + 1}" * order
            derivatives[key] = current

    if 2 in spatial:
        if self.dimension == 1:
            derivatives["laplacian"] = derivatives["dx2"]
        else:
            derivatives["laplacian"] = sum(
                derivatives["d" + f"x{d + 1}" * 2] for d in range(self.dimension)
            )
    return derivatives

    d = self.compute_derivatives(
        model, x, t, temporal_derivatives=[1], spatial_derivatives=[1, 2]
    )
    if self.dimension == 1:
        v_s, v_ss = d["dx"], d["dx2"]
    else:
        v_s = torch.cat([d[f"dx{i + 1}"] for i in range(self.dimension)], dim=1)
        v_ss = torch.cat([d["d" + f"x{i + 1}" * 2] for i in range(self.dimension)], dim=1)
    return (
        d["dt"]
        - 0.5 * self.sigma**2 * torch.sum(x**2 * v_ss, dim=1, keepdim=True)
        - self.r * torch.sum(x * v_s, dim=1, keepdim=True)
        + self.r * d["u"]
    )

    Damped nonlinear pendulum θ'' + b θ' + (g/L) sin θ = 0.

    The angle θ(t) is the network output; the spatial input is a dummy coordinate the solution does not depend on.

    Parameters: g (default 9.81), L (default 1), damping b (default 0).

    Second order in time: the initial loss enforces θ(0) = initial_angle and θ'(0) = velocity (default 0).

    Initial condition: small_angle (alias fixed) with initial_angle (alias value, default 0.5).

    Exact solution: ode — the full nonlinear damped trajectory from the configured initial angle and velocity, integrated with scipy to 1e-11 relative tolerance (there is no elementary closed form; the small-angle formula θ0 cos(ωt) is only an approximation).

    def compute_energy(

    """Mechanical energy per unit mass, ``½ L² θ'² + g L (1 - cos θ)``."""
    d = self.compute_derivatives(model, x, t, temporal_derivatives=[1])
    return 0.5 * self.L**2 * d["dt"] ** 2 + self.g * self.L * (1 - torch.cos(d["u"]))
