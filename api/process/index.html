    Process models¶

    Reference simulators of process units, their CustomPDE forms and the classical estimators they are compared with (pinnrl.process). The falling-film evaporator is the first unit (roadmap v0.7, experiments J01–J06) and the drying droplet, with its drying-rate closure, the second (J07); the spray-dryer tower couples that droplet to its air (J11). The estimators are an extended Kalman filter with a smoother and moving-horizon estimation; Historian turns a plant-historian export into the inputs and sensor records they and the PINN twin take.

    def steady_state(self, nz: int = 201, t: float = 0.0) -> dict[str, np.ndarray]:
    """The steady profile at the inputs' values at time ``t``: an ODE in ``z``.

    Returns ``{"z", "W", "X", "q"}``.
    """
    z = np.linspace(0.0, self.length, nz)
    P, lam, v = self.perimeter, self.latent_heat, self.velocity

    def rhs(_z, s):
        W, X = s
        q = self.heat_flux(X, t)
        return [-P * q / (lam * v), X * P * q / (lam * W * v)]

    sol = solve_ivp(
        rhs,
        (0.0, self.length),
        [_as_fn(self.W_in)(t), _as_fn(self.X_in)(t)],
        t_eval=z,
        method="LSODA",
        rtol=1e-10,
        atol=1e-12,
    )
    if not sol.success:
        raise RuntimeError(f"steady-state solve failed: {sol.message}")
    return {"z": z, "W": sol.y[0], "X": sol.y[1], "q": self.heat_flux(sol.y[1], t)}

    def mol_rhs(self, nz: int = 201) -> Callable[[float, np.ndarray], np.ndarray]:
    """The method-of-lines right-hand side ``ds/dt = f(t, s)`` on ``nz`` axial points.

    The state is ``s = [W_0..W_{nz-1}, X_0..X_{nz-1}]``; first-order upwind
    in ``z``; the inlet entries are overwritten by the feed signals. The
    Jacobian of this function is what the linearized-stability analysis
    (K09) looks at.
    """
    dz = self.length / (nz - 1)
    P, lam, v = self.perimeter, self.latent_heat, self.velocity
    W_in, X_in = _as_fn(self.W_in), _as_fn(self.X_in)

    def rhs(tau, s):
        W, X = s[:nz].copy(), s[nz:].copy()
        W[0], X[0] = W_in(tau), X_in(tau)  # the inlet is a boundary value
        q = self.heat_flux(X, tau)
        dW = np.zeros(nz)
        dX = np.zeros(nz)
        dW[1:] = -v * (W[1:] - W[:-1]) / dz - P * q[1:] / lam
        dX[1:] = -v * (X[1:] - X[:-1]) / dz + X[1:] * P * q[1:] / (lam * W[1:])
        return np.concatenate([dW, dX])

    return rhs

    def reference(
    self,
    t_max: float,
    nz: int = 201,
    nt: int = 201,
    initial: Mapping[str, np.ndarray] | str = "steady",
    method: str = "BDF",
) -> EvaporatorReference:
    """Dynamic solution by the method of lines (first-order upwind, implicit in time).

    Args:
        t_max: Final time, s.
        nz: Axial grid points (including the inlet).
        nt: Output time points on ``[0, t_max]``.
        initial: ``"steady"`` (the steady profile at ``t = 0``) or
            ``{"W": array, "X": array}`` on the ``nz`` grid.
        method: ``solve_ivp`` method; ``BDF`` handles the stiff source.
    """
    z = np.linspace(0.0, self.length, nz)
    t = np.linspace(0.0, t_max, nt)
    W_in, X_in = _as_fn(self.W_in), _as_fn(self.X_in)
    if initial == "steady":
        ss = self.steady_state(nz, t=0.0)
        W0, X0 = ss["W"], ss["X"]
    elif isinstance(initial, str):
        raise ValueError(
            f"initial must be 'steady' or a mapping with 'W' and 'X', not {initial!r}"
        )
    else:
        profiles = cast("Mapping[str, np.ndarray]", initial)
        W0, X0 = np.asarray(profiles["W"], dtype=float), np.asarray(profiles["X"], dtype=float)
    rhs = self.mol_rhs(nz)

    s0 = np.concatenate([W0, X0])
    sol = solve_ivp(rhs, (0.0, t_max), s0, t_eval=t, method=method, rtol=1e-8, atol=1e-11)
    if not sol.success:
        raise RuntimeError(f"evaporator reference failed: {sol.message}")
    W = sol.y[:nz]
    X = sol.y[nz:]
    for k, tau in enumerate(t):  # the inlet row carries the boundary values
        W[0, k], X[0, k] = W_in(tau), X_in(tau)
    q = np.stack([self.heat_flux(X[:, k], tau) for k, tau in enumerate(t)], axis=1)
    return EvaporatorReference(
        W=ReferenceSolution(x=z, t=t, u=W), X=ReferenceSolution(x=z, t=t, u=X), q=q, z=z, t=t
    )

    def fouling_state_space(self, nz: int = 11, steam_bias: bool = False):
    """The interior states ``[W_1..W_{nz-1}, X_1..X_{nz-1}]`` with ``R_f`` as a parameter, for an estimator.

    Returns ``(f, h, s0)``: ``f(t, s, theta)`` the interior dynamics with
    ``theta = [R_f]`` (the other signals from this model's ``W_in``,
    ``X_in`` and ``T_s``), ``h(t, s)`` the outlet records ``[X(L), W(L)]``,
    and ``s0`` the steady interior state at ``t = 0`` with this model's
    ``R_f(0)``. With ``steam_bias`` the parameters are ``theta = [R_f, b]``
    and the steam temperature is ``T_s(t) - b``: the offset of the steam
    record, the input the fouling can be traded against (J04), estimated
    alongside. The Kalman-filter baseline (J05) and the fouling monitor are
    built on it.
    """
    holder = [float(_as_fn(self.R_f)(0.0)), 0.0]
    steam = _as_fn(self.T_s)
    model = replace(self, R_f=lambda t: holder[0], T_s=lambda t: steam(t) - holder[1])
    rhs = model.mol_rhs(nz)
    W_in, X_in = _as_fn(self.W_in), _as_fn(self.X_in)

    def f(t: float, s: np.ndarray, theta: np.ndarray) -> np.ndarray:
        holder[0] = float(theta[0])
        holder[1] = float(theta[1]) if steam_bias else 0.0
        full = np.concatenate([[W_in(t)], s[: nz - 1], [X_in(t)], s[nz - 1 :]])
        d = rhs(t, full)
        return np.concatenate([d[1:nz], d[nz + 1 :]])

    def h(t: float, s: np.ndarray) -> np.ndarray:
        return np.array([s[2 * nz - 3], s[nz - 2]])

    ss = self.steady_state(nz, t=0.0)
    s0 = np.concatenate([ss["W"][1:], ss["X"][1:]])
    return f, h, s0

    def custom_pde(
    self,
    t_max: float,
    *,
    inputs_grid: int = 201,
    unknown_fouling: Mapping[str, Any] | None = None,
    sensors: list | None = None,
    name: str = "Falling-film evaporator",
):
    """The same model as a :class:`~pinnrl.pdes.custom.CustomPDE`.

    Fields ``W`` and ``X`` of ``(z, t)``; the inputs ``W_in``, ``X_in``,
    ``T_s`` and (unless ``unknown_fouling`` is given) ``R_f`` as
    :class:`~pinnrl.pdes.inputs.TimeSeries` sampled on ``inputs_grid``
    points of ``[0, t_max]``; the inlet as one-sided Dirichlet conditions;
    the steady profile at ``t = 0`` as the initial condition; the scales
    of ``W`` and ``X`` from the feed. ``unknown_fouling`` is the spec of an
    :class:`~pinnrl.pdes.unknowns.UnknownFunction` of ``t`` for ``R_f``
    (e.g. ``{"positive": True, "initial": 1e-4, "hidden": [16, 16]}``),
    for the fouling-estimation twin (J03).
    """
    from pinnrl.pdes.custom import CustomPDE
    from pinnrl.pdes.inputs import TimeSeries

    ts = np.linspace(0.0, t_max, inputs_grid)
    series = {
        "W_in": TimeSeries(ts, [_as_fn(self.W_in)(x) for x in ts], name="W_in"),
        "X_in": TimeSeries(ts, [_as_fn(self.X_in)(x) for x in ts], name="X_in"),
        "T_s": TimeSeries(ts, [_as_fn(self.T_s)(x) for x in ts], name="T_s"),
    }
    params = {
        "v": self.velocity,
        "P": self.perimeter,
        "U": self.U,
        "lam": self.latent_heat,
        "T_sat": self.T_sat,
        "k_bpe": self.k_bpe,
    }
    unknown_functions = None
    if unknown_fouling is None:
        series["R_f"] = TimeSeries(ts, [_as_fn(self.R_f)(x) for x in ts], name="R_f")
    else:
        spec = {"inputs": ["t"], "positive": True, "initial": 1e-4, "hidden": [16, 16]}
        spec.update(unknown_fouling)
        unknown_functions = {"R_f": spec}
    flux = "(T_s - (T_sat + k_bpe*X))/(1/U + R_f)"
    ss = self.steady_state(inputs_grid, t=0.0)
    W0 = np.interp
    z_grid = ss["z"]

    def initial(x):
        import torch

        zz = x[:, 0].detach().cpu().numpy()
        W = torch.as_tensor(W0(zz, z_grid, ss["W"]), dtype=x.dtype).reshape(-1, 1)
        X = torch.as_tensor(np.interp(zz, z_grid, ss["X"]), dtype=x.dtype).reshape(-1, 1)
        return torch.cat([W, X], dim=1)

    W_scale = float(_as_fn(self.W_in)(0.0))
    X_scale = float(_as_fn(self.X_in)(0.0))
    return CustomPDE(
        [
            f"W_t + v*W_z + P*{flux}/lam",
            f"X_t + v*X_z - X*P*{flux}/(lam*W)",
        ],
        domain={"z": (0.0, self.length), "t": (0.0, t_max)},
        fields=["W", "X"],
        parameters=params,
        inputs=series,
        unknown_functions=unknown_functions,
        initial_condition=initial,
        boundary_conditions={
            "W": {"dirichlet": {"left": "W_in"}},
            "X": {"dirichlet": {"left": "X_in"}},
        },
        scales={"W": (W_scale, 0.2 * W_scale), "X": (X_scale, 0.5 * X_scale)},
        sensors=sensors,
        name=name,
    )

    """The drying-rate closure read off a mass-and-temperature record, point by point.

    ``f_i = -dX/dt m_s / (h_m A (rho_vs(T_i) - rho_va))`` with the moisture
    and its rate from a Savitzky–Golay filter of the mass record (window
    and polynomial order as given) and the transfer coefficients of this
    model. Returns ``(X_i, f_i)`` for the readings with ``X_i > X_eq +
    margin``, ``f_i`` clipped to ``[0, 1]``: the start for an

    X_rec = np.asarray(mass, dtype=float) / self.m_s - 1.0
    dt = float(times[1] - times[0])

    dX = savgol_filter(X_rec, window, order, deriv=1, delta=dt)

    keep = X > self.X_eq + margin

    # zero volume; such a solve fails or returns NaN, and the misfit is infinite.

    def expressions(self) -> dict[str, str]:
    """The text sub-expressions of the model in the fields ``X`` and ``T`` (and ``f`` for the closure)."""

    """Relative closure of the water and energy balances over the tower.

    ``water``: the water the droplets lost against the water the air
    gained. ``energy``: the change of the air's and the droplets' enthalpy
    plus the wall loss, relative to the heat the air gave up. The model

    the term ``F_s r (c_v - c_w) (T_p - T_0)`` (about 5 % of the latent

    All are zero for the exact solution of the model; the reference closes
    them to the quadrature error of its output grid.

    Fs, G, lam = self.solids_flow, self.air_flow, d.latent_heat

    def enthalpy(k: int) -> float:
        air = G * ((d.cp_air + self.cp_vapour * ref.Y[k]) * (ref.T_a[k] - T0) + lam * ref.Y[k])

        np.trapezoid(self.U_wall * math.pi * self.diameter * (ref.T_a - self.T_ambient), ref.z)

    rate = -np.gradient(ref.X, ref.tau, edge_order=2)

    def step(
    self,
    t_prev: float,
    t: float,
    x: np.ndarray,
    P: np.ndarray,
    y: np.ndarray,
    lower: np.ndarray | None = None,
    upper: np.ndarray | None = None,
) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """One prediction to ``t`` and update with the record ``y``.

    Returns ``(x, P, x_pred, P_pred, F, innovation)``: the posterior, the
    prediction it came from, the transition matrix of the step and
    ``y - h(x_pred)``.
    """
    n = x.size
    dt = t - t_prev
    # linearize at the current estimate, then propagate mean and covariance
    J = finite_difference_jacobian(lambda s, _t=t_prev: self.f(_t, s), x)  # type: ignore[misc]
    F = expm(J * dt) if dt > 0 else np.eye(n)
    x_pred = self.propagate(t_prev, t, x)
    P_pred = F @ P @ F.T + self.Q * dt
    H = finite_difference_jacobian(lambda s, _t=t: self.h(_t, s), x_pred)  # type: ignore[misc]
    y_pred = np.asarray(self.h(t, x_pred), dtype=float).ravel()
    S = H @ P_pred @ H.T + self.R
    K_gain = np.linalg.solve(S, H @ P_pred).T
    x_new = x_pred + K_gain @ (y - y_pred)
    if lower is not None or upper is not None:
        x_new = np.clip(
            x_new,
            lower if lower is not None else -np.inf,
            upper if upper is not None else np.inf,
        )
    IKH = np.eye(n) - K_gain @ H
    P_new = IKH @ P_pred @ IKH.T + K_gain @ self.R @ K_gain.T
    return x_new, 0.5 * (P_new + P_new.T), x_pred, P_pred, F, y - y_pred

    self,
    times: np.ndarray,

    ``lower``/``upper`` clip the posterior mean (bounds on a physical state,
    a fouling resistance that cannot be negative).
    """

    tic = _time.perf_counter()
    times = np.asarray(times, dtype=float)
    records = np.asarray(records, dtype=float).reshape(times.size, -1)

        xs.append(x.copy())
        Ps.append(P.copy())
        x_preds.append(x_pred)
        P_preds.append(P_pred)
        Fs.append(F)
        innov.append(innovation)
        t_prev = t
    result = FilterResult(
        t=times,
        filtered=np.array(xs),
        filtered_cov=np.array(Ps),
        innovations=np.array(innov),
    )

        xs_s, Ps_s = list(xs), list(Ps)
        for k in range(len(times) - 2, -1, -1):
            C = np.linalg.solve(P_preds[k + 1].T, (Ps[k] @ Fs[k + 1].T).T).T
            xs_s[k] = xs[k] + C @ (xs_s[k + 1] - x_preds[k + 1])

    return result

    def time_to_threshold(
    value: float,

    covariance: np.ndarray,
    threshold: float,

    rising: bool | None = None,
    n_samples: int = 4000,
    seed: int = 0,
) -> ThresholdTime:
    """Time until ``value + rate * tau`` reaches ``threshold``, with its uncertainty.

    ``(value, rate)`` is Gaussian with the given ``2 x 2`` covariance (a filter's
    level and rate with their block of the covariance, or a fitted line's);
    the time ``tau = (threshold - value) / rate`` is a ratio of Gaussians, so
    its interval comes from draws rather than a delta-method standard
    deviation: draws whose rate points away from the threshold never reach it
    (``inf``), and draws already past it give zero. The rate is taken as
    constant from now on, which is the trend model's own prediction.
    ``rising`` says from which side the threshold is approached (a fouling
    resistance rises to its limit, a heat-transfer coefficient falls to its
    own); by default it is the side ``value`` is on.
    """
    cov = np.asarray(covariance, dtype=float).reshape(2, 2)
    std = np.sqrt(np.maximum(np.diag(cov), 0.0))
    safe = np.where(std > 0, std, 1.0)
    corr = cov / np.outer(safe, safe)  # sample in units of the standard deviations
    rho = float(np.clip(corr[0, 1], -1.0, 1.0))
    z = np.random.default_rng(seed).standard_normal((n_samples, 2))
    v = value + std[0] * z[:, 0]
    r = rate + std[1] * (rho * z[:, 0] + np.sqrt(1.0 - rho**2) * z[:, 1])
    if rising is None:
        rising = threshold >= value
    direction = 1.0 if rising else -1.0
    gap = threshold - v
    reaches = direction * r > 0
    tau = np.where(reaches, gap / np.where(reaches, r, 1.0), np.inf)
    tau = np.where(direction * gap <= 0, 0.0, tau)  # draws already past the threshold
    # Order statistics, not interpolation: the draws that never arrive are infinite.
    lo, med, hi = np.quantile(tau, [(1 - level) / 2, 0.5, (1 + level) / 2], method="inverted_cdf")
    return ThresholdTime(
        median=float(med),
        lower=float(lo),
        upper=float(hi),
        probability=float(np.isfinite(tau).mean()),
        level=level,
    )

    def run(
    self,
    times: np.ndarray,
    records: np.ndarray,
    x0: np.ndarray,
    P0: np.ndarray,
    t_start: float = 0.0,
    lower: np.ndarray | None = None,
    upper: np.ndarray | None = None,
    noise_lower: np.ndarray | float | None = None,
    noise_upper: np.ndarray | float | None = None,
    stride: int = 1,
) -> FilterResult:
    """Estimate the state at every ``stride``-th record from the window ending there.

    ``lower``/``upper`` bound the state at the start of each window (and
    clip the arrival filter's mean); ``noise_lower``/``noise_upper`` bound
    the disturbance of each state with process noise per record, in the
    state's units (``noise_lower=0`` for a parameter that only grows).
    Until ``horizon`` records have arrived the window grows from the prior.

    Returns:
        A :class:`FilterResult` whose ``t`` holds the solved records,
        ``filtered`` the estimates at the end of each window and
        ``filtered_cov`` their Gauss–Newton covariances.
    """
    tic = time.perf_counter()
    times = np.asarray(times, dtype=float)
    records = np.asarray(records, dtype=float).reshape(times.size, -1)
    x0, P0 = np.asarray(x0, dtype=float), np.asarray(P0, dtype=float)
    n, p, K = x0.size, self.noisy.size, times.size
    lo = np.full(n, -np.inf) if lower is None else np.asarray(lower, dtype=float)
    hi = np.full(n, np.inf) if upper is None else np.asarray(upper, dtype=float)
    w_lo = np.broadcast_to(-np.inf if noise_lower is None else noise_lower, (p,))
    w_hi = np.broadcast_to(np.inf if noise_upper is None else noise_upper, (p,))
    # arrival[i]: the filter's mean and covariance before record i (after record i - 1)
    arrival = [(t_start, x0.copy(), P0.copy())]
    solved_t, xs, Ps, innov = [], [], [], []
    previous = None  # (first record index, start state, states, disturbances) of the last solve
    self.evaluations = 0
    for k in range(K):
        if (k + 1) % stride != 0 and k != K - 1:
            continue
        a = max(0, k + 1 - self.horizon)
        while len(arrival) <= a:
            i = len(arrival) - 1
            t_prev, x, P = arrival[-1]
            x, P, *_ = self.filter.step(t_prev, times[i], x, P, records[i], lower, upper)
            arrival.append((times[i], x, P))
        t_a, mean, cov = arrival[a]
        N = k - a + 1
        z, w = mean.copy(), np.zeros((N, p))
        if previous is not None:  # warm start from the last solution, shifted
            a_prev, z_prev, states_prev, w_prev = previous
            shift = a - a_prev
            if shift == 0:
                z = z_prev
            elif shift <= states_prev.shape[0]:
                z = states_prev[shift - 1]
            keep = w_prev[shift:]
            w[: keep.shape[0]] = keep[:N]
        bounds = (
            np.concatenate([lo, np.tile(w_lo, N)]),
            np.concatenate([hi, np.tile(w_hi, N)]),
        )
        v, states, P_end, innovation, nfev = self._solve(
            t_a,
            times[a : k + 1],
            records[a : k + 1],
            mean,
            cov,
            np.concatenate([z, w.ravel()]),
            bounds,
        )
        self.evaluations += nfev
        previous = (a, v[:n], states, v[n:].reshape(N, p))
        solved_t.append(times[k])
        xs.append(states[-1])
        Ps.append(P_end)
        innov.append(innovation)
    return FilterResult(
        t=np.array(solved_t),
        filtered=np.array(xs),
        filtered_cov=np.array(Ps),
        innovations=np.array(innov),
        time_s=time.perf_counter() - tic,
    )

    def resample(
    self, tag: str, times: Sequence[float] | np.ndarray, aggregate: str = "interp"
) -> np.ndarray:
    """The tag at ``times``.

    ``"interp"`` interpolates linearly between readings (held at the ends);
    ``"mean"`` averages the readings in the interval ending at each time
    (its length the spacing to the previous time), which is what a slower
    record of a fast, noisy tag should be, and falls back to interpolation
    where an interval holds no reading; ``"last"`` takes the latest reading
    at or before each time (a sample-and-hold signal, a lab value).
    """
    t, v = self.readings(tag)
    if t.size == 0:
        raise ValueError(f"tag {tag!r} has no good readings")
    times = np.asarray(times, dtype=float).reshape(-1)
    interp = np.interp(times, t, v)
    if aggregate == "interp":
        return interp
    if aggregate == "last":
        idx = np.searchsorted(t, times, side="right") - 1
        return np.where(idx >= 0, v[np.clip(idx, 0, None)], v[0])
    if aggregate == "mean":
        steps = np.diff(times)
        first = steps[0] if steps.size else 0.0
        starts = times - np.concatenate([[first], steps])
        lo = np.searchsorted(t, starts, side="right")
        hi = np.searchsorted(t, times, side="right")
        csum = np.concatenate([[0.0], np.cumsum(v)])
        count = hi - lo
        with np.errstate(invalid="ignore", divide="ignore"):
            mean = (csum[hi] - csum[lo]) / count
        return np.where(count > 0, mean, interp)
    raise ValueError(f"aggregate {aggregate!r}: use 'interp', 'mean' or 'last'")

    The map is a dict or a YAML file::

        window: [600, 1800]          # optional, s on the historian's clock; shifted to t = 0
        inputs:
          T_s:  {tag: TT205.PV, offset: 273.15}        # degC -> K
          W_in: {tag: FT101.PV, gain: 2.7778e-4}       # kg/h -> kg/s
        sensors:
          - {tag: DT310.PV, name: outlet X, kind: point, location: 10.0,
             expression: X, gain: 0.01, noise: 0.002, every: 5.0, aggregate: mean}

    An input entry takes ``tag`` and optionally ``gain``, ``offset``,
    ``every`` and ``aggregate``. A sensor entry takes ``tag``, the same
    conversion keys, and the keys of the sensor it describes (``kind``:
    ``point``, ``boundary`` or ``integral``; ``expression``, ``location``
    or ``side``, ``noise``, ``delay``, ``weight``, ``scale``, ``name``);
    its timestamps are the tag's readings in the window, or a regular grid
    with ``every``, which starts one period after the window's start.
    Without ``window`` the interval covered by every mapped tag is used.

    Raises:
        ValueError: If the map names a tag the export does not have.
    """
    if not isinstance(tag_map, Mapping):
        import yaml

        tag_map = cast("Mapping[str, Any]", yaml.safe_load(Path(tag_map).read_text()))
    inputs_map = dict(tag_map.get("inputs") or {})
    sensors_map = list(tag_map.get("sensors") or [])
    used = [spec["tag"] for spec in inputs_map.values()] + [s["tag"] for s in sensors_map]
    unknown = sorted({t for t in used if t not in self._tags})
    if unknown:
        raise ValueError(f"the tag map names tags the export does not have: {unknown}")
    if not used:
        raise ValueError("the tag map has neither inputs nor sensors")
    if tag_map.get("window") is not None:
        t0, t1 = (float(v) for v in tag_map["window"])
    else:
        spans = [self.span(t) for t in used]
        t0, t1 = max(s[0] for s in spans), min(s[1] for s in spans)
    if t1 <= t0:
        raise ValueError(f"the mapped tags share no interval (window [{t0}, {t1}])")
    inputs = {
        name: self.series(
            spec["tag"],
            gain=float(spec.get("gain", 1.0)),
            offset=float(spec.get("offset", 0.0)),
            start=t0,
            end=t1,
            every=spec.get("every"),
            aggregate=spec.get("aggregate", "interp"),
            name=name,
        )
        for name, spec in inputs_map.items()
    }
    sensors = []
    for spec in sensors_map:
        spec = dict(spec)
        tag = spec.pop("tag")
        gain, offset = float(spec.pop("gain", 1.0)), float(spec.pop("offset", 0.0))
        every, aggregate = spec.pop("every", None), spec.pop("aggregate", "interp")
        spec.setdefault("kind", "point")
        spec.setdefault("name", tag)
        if every is not None:
            times = np.arange(t0 + float(every), t1 + 1e-9 * max(1.0, abs(t1)), float(every))
            values = self.resample(tag, times, aggregate)
        else:
            t, v = self.readings(tag)
            keep = (t >= t0) & (t <= t1)
            times, values = t[keep], v[keep]
        if times.size == 0:
            raise ValueError(
                f"sensor {spec['name']!r}: tag {tag!r} has no readings in the window"
            )
        sensor = Sensor.from_spec({**spec, "times": (times - t0).tolist()})
        sensor.set_values(gain * values + offset)
        sensors.append(sensor)
    return TwinData(inputs=inputs, sensors=sensors, t_max=t1 - t0, t_start=t0)

    ``names`` selects and orders the sensors; the default is all of them.

        ValueError: If a name is unknown or the sensors do not share their
            timestamps (give them the same ``every`` in the tag map).

    chosen = list(by_name) if names is None else list(names)

        raise ValueError(f"no sensor named {missing}; sensors are {sorted(by_name)}")

        raise ValueError("the tag map defines no sensors")

        raise ValueError(f"sensors {empty} carry no records")

            )

        cast("torch.Tensor", by_name[n].values).detach().cpu().numpy().ravel() for n in chosen

    def monitor_fouling(
    model: FallingFilmEvaporator,
    times: np.ndarray,
    records: np.ndarray,
    *,
    nz: int = 51,
    noise: tuple[float, float] = (0.002, 0.001),
    fouling_std: float = 1e-4,
    rate_std: float = 1e-6,
    rate_walk: float = 1e-9,
    steam_bias_std: float | None = None,
    threshold: float | None = None,
    level: float = 0.9,
) -> FoulingMonitor:
    """Estimate the fouling resistance of ``model`` from outlet records.

    Args:
        model: The evaporator with its recorded signals (``T_s``, ``W_in``,
            ``X_in`` as functions of time); its ``R_f`` at ``t = 0`` is the
            prior mean of the fouling (0 after a cleaning).
        times: Record times, s, increasing, after ``t = 0``.
        records: ``(K, 2)``, the outlet solids fraction and the outlet flow.
        nz: Axial cells of the filter's model. A coarse grid biases the
            estimate by a few percent (3 % at 11 cells, under 0.5 % at 51),
            which matters once the interval is that narrow (J06).
        noise: Standard deviations of the two sensors.
        fouling_std: Prior standard deviation of the fouling at ``t = 0``.
        rate_std: Prior standard deviation of its rate, per second.
        rate_walk: Random-walk intensity of the rate, per second^1.5: how fast
            the rate itself may change. A rate ``r`` that may change by its
            own size over a time ``tau`` corresponds to ``r / sqrt(tau)``.
            Too small a value on a fouling curve that bends gives a narrow
            band around a biased estimate (J06: coverage 7 % instead of 98 %
            on a saturating curve); a larger value is safe and widens the
            band and, more so, the time to clean.
        steam_bias_std: Prior standard deviation of a constant offset of the
            steam record, K. ``None`` trusts the record. A steam offset mimics
            fouling (0.5 K for 30 %, J04), so an interval that ignores a
            plausible offset is too narrow.
        threshold: Fouling at which the tube is cleaned; gives ``time_to_clean``.
        level: Nominal level of the intervals.
    """
    times = np.asarray(times, dtype=float)
    records = np.asarray(records, dtype=float).reshape(times.size, 2)
    with_bias = steam_bias_std is not None
    f, h, s0 = model.fouling_state_space(nz, steam_bias=with_bias)
    n = s0.size
    k_rate, k_bias = n + 1, n + 2
    dim = n + 2 + int(with_bias)

    def dynamics(t: float, x: np.ndarray) -> np.ndarray:
        theta = x[[n, k_bias]] if with_bias else x[n : n + 1]
        out = np.zeros(dim)
        out[:n] = f(t, x[:n], theta)
        out[n] = x[k_rate]
        return out

    x0, P0, Q = np.zeros(dim), np.zeros(dim), np.zeros(dim)
    x0[:n], P0[:n], Q[:n] = s0, 1e-6, 1e-12
    x0[n] = max(float(_as_fn(model.R_f)(0.0)), 0.0)
    P0[n], P0[k_rate], Q[k_rate] = fouling_std**2, rate_std**2, rate_walk**2
    if with_bias:
        assert steam_bias_std is not None  # with_bias is exactly that
        P0[k_bias] = float(steam_bias_std) ** 2
    ekf = ExtendedKalmanFilter(
        dynamics, lambda t, x: h(t, x[:n]), Q=Q, R=np.array([noise[0] ** 2, noise[1] ** 2])
    )
    # The filter's state is not clipped at zero: clipping the fouling alone leaves the
    # states it is correlated with (the steam offset) inconsistent and costs coverage (J06).
    # The reported estimate and band are cut at zero instead.
    res = ekf.run(times, records, x0, np.diag(P0), smooth=True)
    z = float(norm.ppf(0.5 + level / 2))
    std, std_s = res.std(), res.std(smoothed=True)
    smoothed = res.smoothed
    assert smoothed is not None  # the run above asks for the smoother
    predicted = records - res.innovations
    start = max(1, times.size // 10)
    misfit = np.mean((res.innovations[start:] / np.asarray(noise)[None, :]) ** 2, axis=0)
    tt = None
    if threshold is not None:
        idx = [n, k_rate]
        tt = time_to_threshold(
            res.filtered[-1, n],
            res.filtered[-1, k_rate],
            res.filtered_cov[-1][np.ix_(idx, idx)],
            threshold,
            level=level,
            rising=True,
        )
    return FoulingMonitor(
        t=times,
        fouling=np.maximum(res.filtered[:, n], 0.0),
        lower=np.maximum(res.filtered[:, n] - z * std[:, n], 0.0),
        upper=np.maximum(res.filtered[:, n] + z * std[:, n], 0.0),
        smoothed=np.maximum(smoothed[:, n], 0.0),
        smoothed_lower=np.maximum(smoothed[:, n] - z * std_s[:, n], 0.0),
        smoothed_upper=np.maximum(smoothed[:, n] + z * std_s[:, n], 0.0),
        rate=res.filtered[:, k_rate],
        rate_std=std[:, k_rate],
        records=records,
        predicted=predicted,
        misfit=(float(misfit[0]), float(misfit[1])),
        steam_bias=(float(res.filtered[-1, k_bias]), float(std[-1, k_bias])) if with_bias else None,
        threshold=threshold,
        time_to_clean=tt,
        level=level,
    )

    percent solids, kg/h) with sensor noise and dated timestamps. Returns the
    paths of ``evaporator_export.csv`` and ``evaporator_tags.yaml``: the input
    of the dashboard's Process view and a template for a real export.
    """
    import yaml

    folder = Path(folder)
    folder.mkdir(parents=True, exist_ok=True)
    t_max, r_end = 1200.0, 2e-4

    def steam(t: float) -> float:
        return 355.15 + math.sin(2 * math.pi * t / 400.0)

    plant = FallingFilmEvaporator(T_s=steam, R_f=lambda t: r_end * t / t_max)
    ref = plant.reference(t_max=t_max, nz=201, nt=241)
    rng = np.random.default_rng(seed)
    start = 1772344800.0  # 2026-03-01T06:00:00Z
    lines = ["timestamp,TT205.PV,DT310.PV,FT320.PV"]
    for k, t in enumerate(ref.t):
        stamp = datetime.fromtimestamp(start + t, tz=timezone.utc).strftime("%Y-%m-%dT%H:%M:%S")
        solids = 100.0 * (ref.X.u[-1, k] + 0.002 * rng.standard_normal())
        flow = 3600.0 * (ref.W.u[-1, k] + 0.001 * rng.standard_normal())
        lines.append(f"{stamp},{steam(t) - 273.15:.3f},{solids:.4f},{flow:.3f}")
    export = folder / "evaporator_export.csv"
    export.write_text("\n".join(lines) + "\n")
    tag_map = {
        "inputs": {"T_s": {"tag": "TT205.PV", "offset": 273.15}},
        "sensors": [
            {
                "tag": "DT310.PV",
                "name": "outlet solids",
                "location": 10.0,
                "expression": "X",
                "gain": 0.01,
                "noise": 0.002,
            },
            {
                "tag": "FT320.PV",
                "name": "outlet flow",
                "location": 10.0,
                "expression": "W",
                "gain": 1 / 3600.0,
                "noise": 0.001,
            },
        ],
    }
    tags = folder / "evaporator_tags.yaml"
    tags.write_text(yaml.safe_dump(tag_map, sort_keys=False))
    return export, tags
