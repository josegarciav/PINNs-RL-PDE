    def main():
    """Entry point of ``pinnrl-train``: parse flags, build the config and train."""
    # Headless CLI: save figures without a GUI backend. Library imports leave the
    # user's backend alone (inline plots in notebooks keep working).
    import matplotlib

    matplotlib.use("agg")
    logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")
    parser = argparse.ArgumentParser(description="Train a PINN model")
    parser.add_argument(
        "--pde",
        default=None,
        help="PDE config key or display name (e.g. 'heat' or 'Heat Equation')",
    )
    parser.add_argument(
        "--derivatives",
        choices=["auto", "autograd", "native"],
        default="auto",
        help=(
            "'auto' (default) takes the residual's derivatives from the compiled kernel "
            "(pinnrl.native: macOS or Linux, a feed-forward tanh network with or without layer "
            "norm, CPU, derivatives of any order), two to three times as fast per step, and "
            "autograd wherever it does not apply; 'autograd' never uses the kernel; 'native' is "
            "the earlier name of 'auto'."
        ),
    )
    parser.add_argument(
        "--normalize",
        action="store_true",
        help=(
            "Non-dimensionalize the network: inputs mapped from the domain to [-1, 1], "
            "outputs scaled by the 'scales' of a --pde-file spec, loss terms in those units."
        ),
    )
    parser.add_argument(
        "--pde-file",
        default=None,
        help=(
            "YAML file defining your own equation(s): 'equations', 'domain', 'fields', "
            "'parameters', 'initial_condition', 'boundary_conditions', 'exact_solution' "
            "(see the Custom equations guide). Replaces --pde."
        ),
    )
    parser.add_argument("--arch", required=True, help="Architecture (e.g. 'fourier')")
    parser.add_argument("--epochs", type=int, default=None, help="Override number of epochs")
    parser.add_argument("--lr", type=float, default=None, help="Override learning rate")
    parser.add_argument("--batch-size", type=int, default=None, help="Override batch size")
    parser.add_argument(
        "--collocation-points", type=int, default=None, help="Override collocation points"
    )
    parser.add_argument(
        "--boundary-points", type=int, default=None, help="Override boundary points"
    )
    parser.add_argument("--initial-points", type=int, default=None, help="Override initial points")
    parser.add_argument(
        "--rl",
        action="store_true",
        help="Enable the experimental RL bandit collocation sampler (1-D PDEs)",
    )
    parser.add_argument(
        "--collocation-distribution",
        choices=["uniform", "stratified", "residual_based"],
        default=None,
        help=("Collocation point sampling strategy. Overridden to 'adaptive' when --rl is set."),
    )
    parser.add_argument(
        "--optimizer",
        choices=["adam", "lbfgs", "adam_lbfgs"],
        default=None,
        help="Optimizer to use",
    )
    parser.add_argument(
        "--mode",
        choices=["forward", "inverse", "data_only", "data_augmented"],
        default=None,
        help=(
            "Training mode: forward (residual + IC/BC), inverse (residual + "
            "IC/BC + data + trainable parameters), data_only (regression on "
            "observed snapshots only), data_augmented (residual + IC/BC + data)."
        ),
    )
    parser.add_argument(
        "--dataset",
        default=None,
        help="Well dataset name (registered in pinnrl.datasets.WELL_REGISTRY)",
    )
    parser.add_argument(
        "--dataset-split", default="train", help="Well dataset split (train/valid/test)"
    )
    parser.add_argument(
        "--dataset-traj",
        type=int,
        default=1,
        help="Number of WellDataset samples to draw (random, without replacement)",
    )
    parser.add_argument(
        "--dataset-points",
        type=int,
        default=4096,
        help="Number of (x, t, u) points to sub-sample from the loaded slice",
    )
    parser.add_argument("--dataset-seed", type=int, default=0, help="RNG seed for sub-sampling")
    parser.add_argument(
        "--dataset-steps",
        type=int,
        default=None,
        help="Time steps read per trajectory (default: the dataset's; 101, or 11 for 3-D)",
    )
    parser.add_argument(
        "--dataset-base",
        default=None,
        help="Local Well download dir; omit to stream from Hugging Face",
    )
    parser.add_argument(
        "--identify",
        action="append",
        default=[],
        help="Name of a PDE parameter to identify in inverse mode (repeatable)",
    )
    parser.add_argument(
        "--initial-guess",
        action="append",
        default=[],
        help="Initial guess for an identifiable parameter, e.g. 'alpha=0.5' (repeatable)",
    )
    parser.add_argument(
        "--obs-path",
        default=None,
        help="Path to an .npz file with observation data (keys 'x','t','u')",
    )
    parser.add_argument(
        "--obs-noise",
        type=float,
        default=None,
        help="Stddev of Gaussian noise added to synthetic observations",
    )
    parser.add_argument(
        "--obs-points",
        type=int,
        default=None,
        help="Number of synthetic observation points to generate",
    )
    parser.add_argument(
        "--loss-function",
        choices=["mse", "mae", "huber"],
        default=None,
        help="Reduction for residual/BC/IC/data losses",
    )
    parser.add_argument(
        "--huber-delta",
        type=float,
        default=None,
        help="Delta for Huber loss (only used when --loss-function=huber)",
    )
    parser.add_argument(
        "--adaptive-weights",
        choices=["off", "lrw", "rbw"],
        default=None,
        help=(
            "Adaptive loss-term weighting: 'off' for static weights, "
            "'lrw' (gradient-balanced), or 'rbw' (loss-balanced)."
        ),
    )
    parser.add_argument(
        "--lr-scheduler",
        choices=["cosine", "reduce_lr"],
        default=None,
        help=(
            "Learning-rate scheduler: 'cosine' annealing or 'reduce_lr' on a plateau "
            "of the epoch training loss."
        ),
    )
    parser.add_argument(
        "--early-stopping",
        choices=["on", "off"],
        default=None,
        help="Toggle early stopping on validation loss.",
    )
    parser.add_argument(
        "--gradient-clipping",
        type=float,
        default=None,
        help="Max gradient norm; pass 0 to disable clipping.",
    )
    parser.add_argument(
        "--weight-decay",
        type=float,
        default=None,
        help="L2 regularisation coefficient on optimiser weights.",
    )
    parser.add_argument("--config", default=DEFAULT_CONFIG_PATH, help="Path to config.yaml")
    parser.add_argument("--device", default=None, help="Device (cpu, mps, cuda)")
    args = parser.parse_args()

    custom_spec = None
    if args.pde_file:
        with open(args.pde_file) as f:
            custom_spec = yaml.safe_load(f) or {}
        from pinnrl.pdes.custom import resolve_input_paths

        custom_spec = resolve_input_paths(custom_spec, Path(args.pde_file).parent)
    elif args.pde is None:
        parser.error("one of --pde or --pde-file is required")
    else:
        try:
            get_pde_spec(args.pde)
        except KeyError:
            print(f"Unknown PDE: {args.pde}")
            print(f"Available: {', '.join(spec.key for spec in PDE_SPECS)}")
            sys.exit(1)

    with open(args.config) as f:
        yaml_config = yaml.safe_load(f)

    if args.device:
        yaml_config["device"] = args.device

    # Apply hyperparameter overrides from CLI
    if args.lr is not None:
        yaml_config.setdefault("training", {}).setdefault("optimizer_config", {})[
            "learning_rate"
        ] = args.lr
    if args.batch_size is not None:
        yaml_config.setdefault("training", {})["batch_size"] = args.batch_size
    if args.collocation_points is not None:
        yaml_config.setdefault("training", {})["num_collocation_points"] = args.collocation_points
    if args.boundary_points is not None:
        yaml_config.setdefault("training", {})["num_boundary_points"] = args.boundary_points
    if args.initial_points is not None:
        yaml_config.setdefault("training", {})["num_initial_points"] = args.initial_points
    if args.collocation_distribution is not None:
        yaml_config.setdefault("training", {})["collocation_distribution"] = (
            args.collocation_distribution
        )
    if args.optimizer is not None:
        yaml_config.setdefault("training", {})["optimizer"] = args.optimizer
    if args.mode is not None:
        yaml_config.setdefault("training", {})["mode"] = args.mode
    if args.loss_function is not None:
        yaml_config.setdefault("training", {})["loss_function"] = args.loss_function
    if args.huber_delta is not None:
        yaml_config.setdefault("training", {})["huber_delta"] = args.huber_delta
    if args.adaptive_weights is not None:
        aw_block = yaml_config.setdefault("training", {}).setdefault("adaptive_weights", {})
        if args.adaptive_weights == "off":
            aw_block["enabled"] = False
        else:
            aw_block["enabled"] = True
            aw_block["strategy"] = args.adaptive_weights
    if args.lr_scheduler is not None:
        yaml_config.setdefault("training", {})["scheduler_type"] = args.lr_scheduler
    if args.early_stopping is not None:
        es_block = yaml_config.setdefault("training", {}).setdefault("early_stopping", {})
        es_block["enabled"] = args.early_stopping == "on"
    if args.gradient_clipping is not None:
        yaml_config.setdefault("training", {})["gradient_clipping"] = args.gradient_clipping
    if args.weight_decay is not None:
        yaml_config.setdefault("training", {}).setdefault("optimizer_config", {})[
            "weight_decay"
        ] = args.weight_decay

    device = resolve_device(yaml_config.get("device", "cpu"))
    dataset_block = None
    if args.dataset:
        dataset_block = {
            "name": args.dataset,
            "split": args.dataset_split,
            "n_traj": args.dataset_traj,
            "n_points": args.dataset_points,
            "seed": args.dataset_seed,
            "base": args.dataset_base,
            "n_steps": args.dataset_steps,
            "use_defaults": True,
            "mode": args.mode,
        }
    config_dict = build_config_dict(
        yaml_config,
        args.pde,
        args.arch,
        args.rl,
        args.epochs,
        dataset=dataset_block,
        custom=custom_spec,
        normalize=args.normalize,
    )
    config_dict["device"] = str(device)
    config_dict["model"]["derivatives"] = args.derivatives

    # Wire inverse-problem flags through to the PDE config + run-time options.
    if args.identify:
        config_dict["pde"]["trainable_parameters"] = list(args.identify)
    if args.initial_guess:
        guesses = {}
        for spec in args.initial_guess:
            if "=" not in spec:
                print(f"Ignoring malformed --initial-guess '{spec}' (expected name=value)")
                continue
            name, value = spec.split("=", 1)
            try:
                guesses[name.strip()] = float(value)
            except ValueError:
                print(f"Ignoring non-numeric --initial-guess '{spec}'")
        if guesses:
            config_dict["pde"]["parameter_initial_guesses"] = guesses
    if args.obs_path:
        config_dict["pde"]["observation_data"] = {"path": args.obs_path}
    inverse_runtime = {}
    if args.obs_noise is not None:
        inverse_runtime["obs_noise"] = args.obs_noise
    if args.obs_points is not None:
        inverse_runtime["obs_points"] = args.obs_points
    if inverse_runtime:
        config_dict["inverse"] = inverse_runtime

    run_training(config_dict, device)

    def build_parser() -> argparse.ArgumentParser:
    """Construct the argparse tree for ``pinnrl-benchmark``."""
    parser = argparse.ArgumentParser(
        prog="pinnrl-benchmark",
        description=(
            "Run pinnrl reference benchmarks: FDM baselines for heat / wave, "
            "and the collocation-sampling study (uniform, stratified, RAD, RL bandit)."
        ),
    )
    sub = parser.add_subparsers(dest="cmd", required=True)

    fdm = sub.add_parser("fdm", help="Run the bundled FDM solver and report error vs analytical.")
    fdm.add_argument("--pde", choices=("heat", "wave"), required=True)

        "--frequency", type=float, default=1.0, help="Initial-condition frequency (heat only)."

    fdm.add_argument("--nt", type=int, default=4001)
    fdm.add_argument("--csv", default=None, help="Optional path to write CSV output.")
    fdm.set_defaults(func=_fdm_command)

    smp = sub.add_parser(
        "sampling",
        help="Multi-seed comparison of collocation samplers at equal training budget.",
    )
    smp.add_argument("--pdes", nargs="+", choices=STUDY_PDES, default=list(STUDY_PDES))
    smp.add_argument(
        "--strategies", nargs="+", choices=STUDY_STRATEGIES, default=list(STUDY_STRATEGIES)

    smp.set_defaults(func=_sampling_command)

    return parser

    def run_dashboard(port=8050, open_browser=False):
    """Serve the Dash app on the first free port from ``port`` (blocking).

    Args:
        port: Preferred port; up to two higher ports are tried if it is taken.
        open_browser: Open the dashboard in a web browser once the server is listening.
    """
    try:
        from pinnrl.dashboard import app
    except ImportError as e:
        print(f"Error importing dashboard: {e}")
        print("Make sure all dependencies are installed (dash, plotly, etc.)")
        sys.exit(1)
    try:
        server = bind_server(app.server, port)
    except RuntimeError as e:
        print(f"{e}. Please close any running dashboards and try again.")
        sys.exit(1)

    url = f"http://{HOST}:{server.server_port}/"
    print(f"\nStarting PINNs-RL-PDE Dashboard on port {server.server_port}")
    print(f"Open {url} in your browser")
    if open_browser:
        webbrowser.open(url)
    app.enable_dev_tools(debug=False)  # what app.run() sets up before serving
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()

    def bind_server(wsgi_app, port: int, attempts: int = 3):
    """Bind a werkzeug server for ``wsgi_app`` on the first free port from ``port``.

    The real server is bound, not a probe socket, so the port cannot be taken
    between the check and serving, and the check uses the server's own socket
    options (werkzeug sets ``SO_REUSEADDR``, so a port in ``TIME_WAIT`` counts
    as free).

    Args:
        wsgi_app: The WSGI application to serve (``app.server`` for Dash).
        port: Preferred port; ``port + 1, ...`` are tried next.
        attempts: How many ports to try.

    Returns:
        A bound, listening ``werkzeug.serving.BaseWSGIServer``; call
        ``serve_forever()`` on it.

    Raises:
        RuntimeError: If every candidate port is taken.
    """
    from werkzeug.serving import make_server

    for candidate in range(port, port + attempts):
        # On a bind error werkzeug prints advice to stderr and calls sys.exit(1).
        with contextlib.redirect_stderr(io.StringIO()):
            try:
                return make_server(HOST, candidate, wsgi_app, threaded=True)
            except (OSError, SystemExit):
                pass
        print(f"Port {candidate} is in use.")
    raise RuntimeError(f"No free port in {port}-{port + attempts - 1}")
